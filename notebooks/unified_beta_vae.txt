# Cell 1: Import
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.utils.data as data
import numpy as np
import psycopg2
import json
from datetime import datetime

# Cell 2: DB Connection
conn = psycopg2.connect(
    dbname='postgres', user='postgres', password='postgres', host='localhost', port=5432
)
cursor = conn.cursor()

# Cell 3: Load Data (wavelet + dct)
def load_vectors(table_name: str, dim_limit: int = 512):
    query = f"""
        SELECT embedding FROM {table_name}
        WHERE embedding IS NOT NULL AND original_dim <= {dim_limit}
    """
    cursor.execute(query)
    rows = cursor.fetchall()
    vectors = [np.array(json.loads(row[0])) for row in rows]
    return vectors

wavelet_vectors = load_vectors('wavelet_vector')
dct_vectors = load_vectors('dct_vector')

all_vectors = wavelet_vectors + dct_vectors
original_dims = [len(vec) for vec in all_vectors]
max_dim = max(original_dims)

# Cell 4: Pad Vectors
padded_vectors = [np.pad(vec, (0, max_dim - len(vec))) for vec in all_vectors]
X = torch.tensor(padded_vectors, dtype=torch.float32)

# Cell 5: VAE Model
class BetaVAE(nn.Module):
    def __init__(self, input_dim, latent_dim=128, beta=4.0):
        super(BetaVAE, self).__init__()
        self.fc1 = nn.Linear(input_dim, 512)
        self.fc21 = nn.Linear(512, latent_dim)  # mu
        self.fc22 = nn.Linear(512, latent_dim)  # logvar
        self.fc3 = nn.Linear(latent_dim, 512)
        self.fc4 = nn.Linear(512, input_dim)
        self.beta = beta

    def encode(self, x):
        h1 = F.relu(self.fc1(x))
        return self.fc21(h1), self.fc22(h1)

    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        h3 = F.relu(self.fc3(z))
        return self.fc4(h3)

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        return self.decode(z), mu, logvar

    def loss_function(self, recon_x, x, mu, logvar, mask):
        recon_loss = ((recon_x - x) ** 2 * mask).sum() / mask.sum()
        kld = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp()) / x.size(0)
        return recon_loss + self.beta * kld

# Cell 6: Training Setup
input_dim = max_dim
latent_dim = 128
vae = BetaVAE(input_dim=input_dim, latent_dim=latent_dim)
optimizer = torch.optim.Adam(vae.parameters(), lr=1e-3)

mask = torch.tensor([[1.0] * d + [0.0] * (max_dim - d) for d in original_dims])

epochs = 50
batch_size = 64
dataset = data.TensorDataset(X, mask)
dataloader = data.DataLoader(dataset, batch_size=batch_size, shuffle=True)

# Cell 7: Training Loop
for epoch in range(epochs):
    vae.train()
    total_loss = 0
    for x_batch, m_batch in dataloader:
        optimizer.zero_grad()
        recon_batch, mu, logvar = vae(x_batch)
        loss = vae.loss_function(recon_batch, x_batch, mu, logvar, m_batch)
        loss.backward()
        optimizer.step()
        total_loss += loss.item()
    print(f"Epoch {epoch+1}, Loss: {total_loss:.4f}")

# Cell 8: Extract Embeddings
def extract_latent(X):
    vae.eval()
    with torch.no_grad():
        mu, _ = vae.encode(X)
    return mu

embeddings = extract_latent(X).numpy()
